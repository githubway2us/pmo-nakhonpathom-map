const UPLOAD_DIR = path.join(__dirname, 'uploads');
fs.mkdirSync(UPLOAD_DIR, { recursive: true });

const db = new Database(path.join(__dirname, 'data.db'));
db.pragma('journal_mode = WAL');
db.exec(`
CREATE TABLE IF NOT EXISTS users (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  username TEXT UNIQUE NOT NULL,
  password TEXT NOT NULL,
  first_name TEXT, last_name TEXT, nickname TEXT,
  affiliation TEXT, affiliation_other TEXT,
  phone TEXT,
  avatar TEXT,
  facebook TEXT, tiktok TEXT, line_id TEXT, website TEXT,
  role TEXT NOT NULL DEFAULT 'user',
  status TEXT NOT NULL DEFAULT 'pending',
  created_at TEXT DEFAULT (datetime('now','localtime'))
);
CREATE TABLE IF NOT EXISTS points (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  user_id INTEGER NOT NULL,
  lat REAL NOT NULL, lng REAL NOT NULL,
  case_name TEXT NOT NULL,
  case_type TEXT NOT NULL, case_type_other TEXT,
  caretaker_name TEXT, caretaker_phone TEXT,
  address TEXT, note TEXT,
  status TEXT NOT NULL DEFAULT 'pending',
  reject_reason TEXT,
  created_at TEXT DEFAULT (datetime('now','localtime')),
  FOREIGN KEY(user_id) REFERENCES users(id)
);
CREATE TABLE IF NOT EXISTS point_photos (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  point_id INTEGER NOT NULL,
  file TEXT NOT NULL,
  FOREIGN KEY(point_id) REFERENCES points(id) ON DELETE CASCADE
);
CREATE INDEX IF NOT EXISTS idx_points_status ON points(status);
CREATE INDEX IF NOT EXISTS idx_points_user ON points(user_id);
`);

const storage = multer.diskStorage({
  destination: (_, __, cb) => cb(null, UPLOAD_DIR),
  filename: (_, file, cb) =>
    cb(null, Date.now() + '-' + Math.round(Math.random() * 1e9) + path.extname(file.originalname).toLowerCase())
});
const upload = multer({
  storage,

function sign(u) { return jwt.sign({ id: u.id, role: u.role }, SECRET, { expiresIn: '30d' }); }

function auth(req, res, next) {
  const t = req.cookies.token;
  if (!t) return res.status(401).json({ error: 'กรุณาเข้าสู่ระบบ' });
  try {
    const p = jwt.verify(t, SECRET);
    const u = db.prepare('SELECT * FROM users WHERE id=?').get(p.id);
    if (!u) return res.status(401).json({ error: 'ไม่พบผู้ใช้' });
    req.user = u; next();
  } catch { res.status(401).json({ error: 'เซสชันหมดอายุ' }); }
}
const approved = (req, res, next) =>
  req.user.status === 'approved' ? next() : res.status(403).json({ error: 'บัญชียังไม่ได้รับการอนุมัติ' });
const adminOnly = (req, res, next) =>
  req.user.role === 'admin' ? next() : res.status(403).json({ error: 'เฉพาะผู้ดูแลระบบ' });

const safeUser = u => ({
  id: u.id, username: u.username, firstName: u.first_name, lastName: u.last_name,
  nickname: u.nickname, affiliation: u.affiliation, affiliationOther: u.affiliation_other,
  phone: u.phone, avatar: u.avatar, facebook: u.facebook, tiktok: u.tiktok,
  lineId: u.line_id, website: u.website, role: u.role, status: u.status, createdAt: u.created_at
});

function removeFile(rel) {
  if (!rel) return;
  const f = path.join(__dirname, rel.replace(/^\//, ''));
  if (f.startsWith(UPLOAD_DIR)) fs.unlink(f, () => {});
}

// จำกัดจำนวนครั้งการล็อกอินที่ผิด (กันเดารหัสผ่าน)
const loginTries = new Map();
setInterval(() => {
  const now = Date.now();
  for (const [k, v] of loginTries) if (now - v.at > 30 * 60000) loginTries.delete(k);
}, 10 * 60000).unref();

app.post('/api/login', (req, res) => {
  const { username, password } = req.body || {};
  const key = req.ip + '|' + (username || '');
  const t = loginTries.get(key) || { n: 0, at: 0 };
  if (t.n >= 8 && Date.now() - t.at < 10 * 60000)
    return res.status(429).json({ error: 'พยายามเข้าสู่ระบบบ่อยเกินไป กรุณาลองใหม่ใน 10 นาที' });

  const u = db.prepare('SELECT * FROM users WHERE username=?').get(username || '');
  if (!u || !bcrypt.compareSync(password || '', u.password)) {
    loginTries.set(key, { n: t.n + 1, at: Date.now() });
    return res.status(401).json({ error: 'ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง' });
  }
  if (u.status === 'pending') return res.status(403).json({ error: 'บัญชีรออนุมัติจากผู้ดูแลระบบ' });
  if (u.status === 'rejected') return res.status(403).json({ error: 'บัญชีถูกปฏิเสธ กรุณาติดต่อผู้ดูแล' });

  loginTries.delete(key);
  res.cookie('token', sign(u), {
    httpOnly: true, sameSite: 'lax', secure: PROD, maxAge: 30 * 864e5
  });
  res.json({ ok: true, user: safeUser(u) });
});

/* ---------- Points ---------- */
const pointRow = `SELECT p.*, u.first_name, u.last_name, u.nickname, u.affiliation,
  u.affiliation_other, u.phone AS reporter_phone, u.avatar FROM points p JOIN users u ON u.id=p.user_id`;

function shape(r) {
  if (!r) return null;
  return {
    id: r.id,
    reporterId: r.user_id,
    lat: r.lat, lng: r.lng, caseName: r.case_name,
    caseType: r.case_type, caseTypeOther: r.case_type_other,
    caretakerName: r.caretaker_name, caretakerPhone: r.caretaker_phone,
    address: r.address, note: r.note, status: r.status, rejectReason: r.reject_reason,
    createdAt: r.created_at,
    reporter: {
      name: `${r.first_name} ${r.last_name}` + (r.nickname ? ` (${r.nickname})` : ''),
      affiliation: r.affiliation === 'อื่นๆ' ? r.affiliation_other : r.affiliation,
      phone: r.reporter_phone, avatar: r.avatar
    },
    photos: db.prepare('SELECT file FROM point_photos WHERE point_id=?').all(r.id).map(x => x.file)
  };
}

app.get('/api/points', auth, approved, (req, res) => {
  const rows = db.prepare(`${pointRow} WHERE p.status='approved' OR p.user_id=? ORDER BY p.id DESC`).all(req.user.id);
  res.json(rows.map(shape));
});

app.get('/api/points/:id', auth, approved, (req, res) => {
  const r = db.prepare(`${pointRow} WHERE p.id=?`).get(req.params.id);
  if (!r) return res.status(404).json({ error: 'ไม่พบข้อมูล' });
  if (r.status !== 'approved' && r.user_id !== req.user.id && req.user.role !== 'admin')
    return res.status(403).json({ error: 'ไม่มีสิทธิ์' });
  res.json(shape(r));
});

app.post('/api/points', auth, approved, upload.array('photos', 6), (req, res) => {
  const b = req.body;
  const lat = parseFloat(b.lat), lng = parseFloat(b.lng);
  if (Number.isNaN(lat) || Number.isNaN(lng) || Math.abs(lat) > 90 || Math.abs(lng) > 180)
    return res.status(400).json({ error: 'พิกัดไม่ถูกต้อง' });
  if (!b.case_name || !b.case_type) return res.status(400).json({ error: 'กรอกชื่อเคสและลักษณะเคส' });

  const info = db.prepare(`INSERT INTO points (user_id,lat,lng,case_name,case_type,case_type_other,
    caretaker_name,caretaker_phone,address,note) VALUES (?,?,?,?,?,?,?,?,?,?)`).run(
    req.user.id, lat, lng, b.case_name, b.case_type, b.case_type_other || '',
    b.caretaker_name || '', b.caretaker_phone || '', b.address || '', b.note || '');

  const ins = db.prepare('INSERT INTO point_photos (point_id,file) VALUES (?,?)');
  (req.files || []).forEach(f => ins.run(info.lastInsertRowid, '/uploads/' + f.filename));

  const point = shape(db.prepare(`${pointRow} WHERE p.id=?`).get(info.lastInsertRowid));
  io.to('admins').emit('point:pending', point);
  io.to('user:' + req.user.id).emit('point:mine', point);
  res.json({ ok: true, message: 'บันทึกพิกัดสำเร็จ รอผู้ดูแลระบบอนุมัติ', point });
});

app.post('/api/admin/users/:id/status', auth, adminOnly, (req, res) => {
  const { status } = req.body;
  if (!['approved', 'rejected', 'pending'].includes(status))
    return res.status(400).json({ error: 'สถานะไม่ถูกต้อง' });
  const u = db.prepare('SELECT id,role FROM users WHERE id=?').get(req.params.id);
  if (!u) return res.status(404).json({ error: 'ไม่พบสมาชิกนี้' });
  if (u.role === 'admin' && status !== 'approved')
    return res.status(400).json({ error: 'ไม่สามารถระงับบัญชีผู้ดูแลระบบได้' });
  db.prepare('UPDATE users SET status=? WHERE id=?').run(status, u.id);
  io.to('user:' + u.id).emit('user:status', { status });
  res.json({ ok: true });
});

app.delete('/api/admin/users/:id', auth, adminOnly, (req, res) => {
  const id = Number(req.params.id);
  if (id === req.user.id) return res.status(400).json({ error: 'ลบบัญชีตัวเองไม่ได้' });
  const u = db.prepare('SELECT id,avatar FROM users WHERE id=?').get(id);
  if (!u) return res.status(404).json({ error: 'ไม่พบสมาชิกนี้' });

  const pts = db.prepare('SELECT id FROM points WHERE user_id=?').all(id);
  const photos = db.prepare(`SELECT ph.file FROM point_photos ph
    JOIN points p ON p.id=ph.point_id WHERE p.user_id=?`).all(id);

  const tx = db.transaction(uid => {
    db.prepare(`DELETE FROM point_photos WHERE point_id IN (SELECT id FROM points WHERE user_id=?)`).run(uid);
    db.prepare('DELETE FROM points WHERE user_id=?').run(uid);
    db.prepare('DELETE FROM users WHERE id=?').run(uid);
  });
  tx(id);

  photos.forEach(p => removeFile(p.file));
  removeFile(u.avatar);
  pts.forEach(p => io.emit('point:removed', { id: p.id }));
  io.to('user:' + id).emit('user:status', { status: 'rejected' });
  res.json({ ok: true, removedPoints: pts.length });
});

app.get('/api/admin/points', auth, adminOnly, (req, res) => {

app.post('/api/admin/points/:id/status', auth, adminOnly, (req, res) => {
  const { status, reason } = req.body;
  if (!['approved', 'rejected', 'pending'].includes(status))
    return res.status(400).json({ error: 'สถานะไม่ถูกต้อง' });
  const exists = db.prepare('SELECT 1 FROM points WHERE id=?').get(req.params.id);
  if (!exists) return res.status(404).json({ error: 'ไม่พบพิกัดนี้' });

  db.prepare('UPDATE points SET status=?, reject_reason=? WHERE id=?')
    .run(status, reason || '', req.params.id);
  const point = shape(db.prepare(`${pointRow} WHERE p.id=?`).get(req.params.id));

  if (status === 'approved') io.emit('point:approved', point);
  else io.emit('point:removed', { id: point.id });
  io.to('user:' + point.reporterId).emit('point:mine', point);
  res.json({ ok: true, point });
});

app.delete('/api/admin/points/:id', auth, adminOnly, (req, res) => {
  const id = Number(req.params.id);
  const exists = db.prepare('SELECT 1 FROM points WHERE id=?').get(id);
  if (!exists) return res.status(404).json({ error: 'ไม่พบพิกัดนี้' });
  const photos = db.prepare('SELECT file FROM point_photos WHERE point_id=?').all(id);
  db.prepare('DELETE FROM point_photos WHERE point_id=?').run(id);
  db.prepare('DELETE FROM points WHERE id=?').run(id);
  photos.forEach(p => removeFile(p.file));
  io.emit('point:removed', { id });
  res.json({ ok: true });
});

/* ---------- Socket.IO (เรียลไทม์) ---------- */
io.on('connection', socket => {
  const m = (socket.handshake.headers.cookie || '').match(/token=([^;]+)/);
  if (!m) return socket.disconnect(true);
  try {
    const p = jwt.verify(decodeURIComponent(m[1]), SECRET);
    const u = db.prepare('SELECT role,status FROM users WHERE id=?').get(p.id);
    if (!u || u.status !== 'approved') return socket.disconnect(true);
    socket.join('user:' + p.id);
    if (u.role === 'admin') socket.join('admins');
  } catch { socket.disconnect(true); }
});

# ระบบบันทึกพิกัดผู้ประสบปัญหาทางสังคม
สำนักงานพัฒนาสังคมและความมั่นคงของมนุษย์จังหวัดนครปฐม
พัฒนาโดย จนท.ระบบงานคอมพิวเตอร์

## ติดตั้ง
1. node setup.js
2. npm install
3. npm start
4. เปิด http://localhost:3000

## บัญชีแอดมินเริ่มต้น
username: admin
password: admin123
(เปลี่ยนรหัสผ่านทันทีหลังเข้าใช้งานครั้งแรก ที่เมนูโปรไฟล์)

## หมายเหตุสำคัญ
- GPS บนเบราว์เซอร์ทำงานเฉพาะ HTTPS หรือ localhost เท่านั้น
- ตั้งค่าก่อนขึ้นโปรดักชัน:
  NODE_ENV=production
  JWT_SECRET=ใส่ค่าสุ่มยาว ๆ ของคุณเอง
- สำรองข้อมูล: คัดลอกไฟล์ data.db และโฟลเดอร์ uploads/

## โครงสร้าง
- server.js            เซิร์ฟเวอร์ API + Socket.IO
- public/index.html    หน้าเข้าสู่ระบบ / สมัครสมาชิก
- public/app.html      แผนที่ + บันทึกพิกัด + โปรไฟล์
- public/admin.html    หลังบ้านผู้ดูแลระบบ
- public/style.css     ธีมหลัก
- data.db              ฐานข้อมูล SQLite (สร้างอัตโนมัติ)
- uploads/             ไฟล์รูปภาพ (สร้างอัตโนมัติ)
